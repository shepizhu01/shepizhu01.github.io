const SAVE_KEY = "xuanhuang_save";
function saveGame(){
  const saveData = {
    player:{
      hp:player.hp,
      maxHp:player.maxHp,
      attackBonus:player.attackBonus,
      hpBonus:player.hpBonus,
      shieldBonus:player.shieldBonus,
      stone:player.stone
    },
    currentBoss,
    scene
  };
  localStorage.setItem(SAVE_KEY, JSON.stringify(saveData));
  alert("✅ 存档成功！");
}
function loadGame(){
  const raw = localStorage.getItem(SAVE_KEY);
  if(!raw){ alert("⚠️ 没有存档"); return; }
  const sd = JSON.parse(raw);
  player.hp = sd.player.hp;
  player.maxHp = sd.player.maxHp;
  player.attackBonus = sd.player.attackBonus;
  player.hpBonus = sd.player.hpBonus;
  player.shieldBonus = sd.player.shieldBonus;
  player.stone = sd.player.stone;
  currentBoss = sd.currentBoss;
  scene = sd.scene;
  resetBoss();
  alert("✅ 读取成功！");
}
function exportSave(){
  const raw = localStorage.getItem(SAVE_KEY);
  if(!raw){ alert("⚠️ 无存档"); return; }
  const blob = new Blob([raw], {type:"application/json"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = "玄荒秘境_存档.json";
  a.click();
  URL.revokeObjectURL(url);
}
function importSave(file){
  const reader = new FileReader();
  reader.onload = e=>{
    try{
      JSON.parse(e.target.result);
      localStorage.setItem(SAVE_KEY, e.target.result);
      loadGame();
    }catch(err){ alert("存档文件损坏"); }
  };
  reader.readAsText(file);
}

};

function resetBoss(){
  playerBullets = [];
  bossBullets = [];
  player.hp = player.maxHp + player.hpBonus;
  player.x = 180;
  player.y = H/2;
  if(currentBoss === "qiongqi"){
    qiongqi.hp = qiongqi.maxHp;
    qiongqi.timer = 0;
  }else{
    taotie.hp = taotie.maxHp;
    taotie.timer = 0;
  }
}

function rectHit(a,b){
  return a.x < b.x+b.w && a.x+a.w> b.x && a.y < b.y+b.h && a.y+a.h> b.y;
}

function update(){
  if(shopOpen || eventOpen) return;
  if(scene === "map"){
    if(keys.W && mapPlayer.y>80) mapPlayer.y -= 3.6;
    if(keys.S && mapPlayer.y<H-80) mapPlayer.y += 3.6;
    if(keys.A && mapPlayer.x>80) mapPlayer.x -= 3.6;
    if(keys.D && mapPlayer.x<W-80) mapPlayer.x += 3.6;
    return;
  }
  if(keys.W && player.y>10) player.y -= player.speed;
  if(keys.S && player.y+player.h < H-10) player.y += player.speed;
  if(keys.A && player.x>10) player.x -= player.speed;
  if(keys.D && player.x+player.w < W/2) player.x += player.speed;

  if(player.atkCd>0) player.atkCd--;
  if(player.shieldCd>0) player.shieldCd--;
  if(player.shieldDur>0){
    player.shieldDur--;
  }else{
    player.shieldActive = false;
  }

  if(currentBoss === "qiongqi"){
    qiongqi.timer++;
    if(qiongqi.timer >= qiongqi.shootGap){
      qiongqi.timer = 0;
      const baseAng = Math.PI;
      for(let i=-3; i<=3; i++){
        const ang = baseAng + i * 0.16;
        bossBullets.push({
          type:"fire",
          x:qiongqi.x, y:qiongqi.y + qiongqi.h/2,
          r:18, vx:Math.cos(ang)*5.2, vy:Math.sin(ang)*5.2
        })
      }
    }
  }else{
    taotie.timer++;
    if(taotie.timer >= taotie.shootGap){
      taotie.timer = 0;
      const baseAng = Math.PI;
      for(let i=-4; i<=4; i++){
        const ang = baseAng + i * 0.14;
        bossBullets.push({
          type:"poison",
          x:taotie.x, y:taotie.y + taotie.h/2,
          r:22, vx:Math.cos(ang)*4.5, vy:Math.sin(ang)*4.5
        })
      }
    }
  }

  for(let i=playerBullets.length-1;i>=0;i--){
    const b = playerBullets[i];
    b.x += b.vx;
    let hitBoss = false;
    if(currentBoss === "qiongqi"){
      if(rectHit(b,qiongqi)){
        qiongqi.hp -= b.dmg; hitBoss = true;
      }
    }else{
      if(rectHit(b,taotie)){
        taotie.hp -= b.dmg; hitBoss = true;
      }
    }
    if(hitBoss){
      playerBullets.splice(i,1);
      continue;
    }
    if(b.x > W) playerBullets.splice(i,1);
  }

  for(let i=bossBullets.length-1;i>=0;i--){
    const b = bossBullets[i];
    b.x += b.vx; b.y += b.vy;
    if(!player.shieldActive){
      if(Math.hypot(b.x-(player.x+player.w/2), b.y-(player.y+player.h/2)) < 44){
        player.hp -= 8;
        bossBullets.splice(i,1);
        continue;
      }
    }else{
      if(Math.hypot(b.x-(player.x+player.w/2), b.y-(player.y+player.h/2)) < 44){
        bossBullets.splice(i,1);
        continue;
      }
    }
    if(b.x<0||b.x>W||b.y<0||b.y>H) bossBullets.splice(i,1);
  }

  if(currentBoss === "qiongqi" && qiongqi.hp <= 0){
    inBattle = false;
    bossBullets = [];
  }
  if(currentBoss === "taotie" && taotie.hp <= 0){
    inBattle = false;
    bossBullets = [];
  }
  if(player.hp <=0) resetBoss();
}

function draw(){

    if(allLoaded && res.mapBg.complete){
      ctx.drawImage(res.mapBg,0,0,W,H);
    }else{
      const grad = ctx.createLinearGradient(0,0,W,H);
      grad.addColorStop(0,"#101814");
      grad.addColorStop(1,"#050806");
      ctx.fillStyle = grad;
      ctx.fillRect(0,0,W,H);
    }
    for(let n of mapNodes){
      ctx.beginPath();
      ctx.arc(n.x, n.y, 28,0,Math.PI*2);
      if(n.type==="boss") ctx.fillStyle="#993322";
      if(n.type==="shop") ctx.fillStyle="#e6b860";
      if(n.type==="event") ctx.fillStyle="#449966";
      ctx.fill();
      ctx.fillStyle="#fff";
      ctx.font="13px SimHei";
      ctx.textAlign="center";
      ctx.fillText(n.name, n.x, n.y+42);
    }
    ctx.beginPath();
    ctx.arc(mapPlayer.x, mapPlayer.y, 24,0,Math.PI*2);
    ctx.fillStyle="#bbddee";
    ctx.fill();
    ctx.fillStyle="#ffe270";
    ctx.font="16px SimHei";
    ctx.textAlign="left";
    ctx.fillText("WASD移动｜空格交互｜B返回战场", 20, 36);
    return;
  }

  if(currentBoss === "qiongqi"){
    if(allLoaded && res.bgLava.complete){
      ctx.drawImage(res.bgLava,0,0,W,H);
    }else{
      const grad = ctx.createLinearGradient(0,0,0,H);
      grad.addColorStop(0,"#28140c");
      grad.addColorStop(1,"#120704");
      ctx.fillStyle = grad;
      ctx.fillRect(0,0,W,H);
    }
  }else{
    if(allLoaded && res.bgSwamp.complete){
      ctx.drawImage(res.bgSwamp,0,0,W,H);
    }else{
      const grad = ctx.createLinearGradient(0,0,0,H);
      grad.addColorStop(0,"#0f1f16");
      grad.addColorStop(1,"#060d09");
      ctx.fillStyle = grad;
      ctx.fillRect(0,0,W,H);
    }
  }

  if(allLoaded && res.merchant.complete){
    ctx.drawImage(res.merchant, merchant.x, merchant.y, merchant.w, merchant.h);
  }else{
    ctx.fillStyle="#776044";
    ctx.fillRect(merchant.x,merchant.y,merchant.w,merchant.h);
  }
  ctx.fillStyle="#e8c880";
  ctx.font="14px SimHei";
  ctx.fillText("秘境商人", merchant.x, merchant.y - 6);

  if(allLoaded && res.player.complete){
    ctx.drawImage(res.player, player.x, player.y, player.w, player.h);
  }else{
    ctx.fillStyle="#cceeff";
    ctx.fillRect(player.x,player.y,player.w,player.h);
  }
  if(player.shieldActive && allLoaded && res.shield.complete){
    ctx.drawImage(res.shield, player.x-12, player.y-12, player.w+24, player.h+24);
  }else if(player.shieldActive){
    ctx.beginPath();
    ctx.arc(player.x+player.w/2, player.y+player.h/2, 54,0,Math.PI*2);
    ctx.strokeStyle="#77ddff";
    ctx.lineWidth=3;
    ctx.stroke();
  }

  if(currentBoss === "qiongqi"){
    if(allLoaded && res.bossQiongqi.complete){
      ctx.drawImage(res.bossQiongqi, qiongqi.x, qiongqi.y, qiongqi.w, qiongqi.h);
    }else{
      ctx.fillStyle="#993322";
      ctx.fillRect(qiongqi.x,qiongqi.y,qiongqi.w,qiongqi.h);
    }
  }else{
    if(allLoaded && res.bossTaotie.complete){
      ctx.drawImage(res.bossTaotie, taotie.x, taotie.y, taotie.w, taotie.h);
    }else{
      ctx.fillStyle="#337744";
      ctx.fillRect(taotie.x,taotie.y,taotie.w,taotie.h);
    }
  }

  playerBullets.forEach(b=>{
    if(allLoaded && res.swordBullet.complete){
      ctx.drawImage(res.swordBullet, b.x,b.y,b.w,b.h);
    }else{
      ctx.fillStyle="#aaddff";
      ctx.fillRect(b.x,b.y,b.w,b.h);
    }
  })
  bossBullets.forEach(b=>{
    if(b.type === "fire"){
      if(allLoaded && res.fireBullet.complete){
        ctx.drawImage(res.fireBullet, b.x-b.r, b.y-b.r, b.r*2, b.r*2);
      }else{
        ctx.beginPath();
        ctx.arc(b.x,b.y,b.r,0,Math.PI*2);
        ctx.fillStyle="#ff6622";
        ctx.fill();
      }
    }else{
      if(allLoaded && res.poisonBullet.complete){
        ctx.drawImage(res.poisonBullet, b.x-b.r, b.y-b.r, b.r*2, b.r*2);
      }else{
        ctx.beginPath();
        ctx.arc(b.x,b.y,b.r,0,Math.PI*2);
        ctx.fillStyle="#44bb66";
        ctx.fill();
      }
    }
  })

  ctx.fillStyle="#222";
  ctx.fillRect(player.x, player.y-14, player.w,8);
  ctx.fillStyle="#3ee077";
  ctx.fillRect(player.x, player.y-14, player.w*((player.hp)/(player.maxHp+player.hpBonus)),8);

  if(currentBoss === "qiongqi"){
    ctx.fillStyle="#222";
    ctx.fillRect(qiongqi.x, qiongqi.y-18, qiongqi.w,10);
    ctx.fillStyle="#ee4433";
    ctx.fillRect(qiongqi.x, qiongqi.y-18, qiongqi.w*(qiongqi.hp/qiongqi.maxHp),10);
  }else{
    ctx.fillStyle="#222";
    ctx.fillRect(taotie.x, taotie.y-18, taotie.w,10);
    ctx.fillStyle="#44bb66";
    ctx.fillRect(taotie.x, taotie.y-18, taotie.w*(taotie.hp/taotie.maxHp),10);
  }
  ctx.fillStyle="#f8c048";
  ctx.font="16px SimHei";
  ctx.fillText(`灵石：${player.stone}`, 20, H - 20);
  if(!inBattle){
    ctx.fillStyle="#ffdd77";
    ctx.font="18px SimHei";
    ctx.fillText("BOSS已击退！按M找商人，B打开大地图", W/2 - 240, 40);
  }
}

function loop(){
  update();
  draw();
  requestAnimationFrame(loop);
}
loop();
